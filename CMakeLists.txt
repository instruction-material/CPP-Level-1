cmake_minimum_required(VERSION 3.20)
project(CPP_Level_1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Each directory is a separate program. Never link multiple main functions.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/COURSE_SOURCE_MANIFEST.md" active_rows
     REGEX "^\\| `CPPF[^`]+` \\|$")
foreach(row IN LISTS active_rows)
    string(REGEX REPLACE "^\\| `([^`]+)` \\|$" "\\1" folder "${row}")
    foreach(part IN ITEMS root starter solution)
        set(directory "${CMAKE_CURRENT_SOURCE_DIR}/${folder}")
        if(NOT part STREQUAL "root")
            string(APPEND directory "/${part}")
        endif()
        file(GLOB sources CONFIGURE_DEPENDS "${directory}/*.cpp")
        if(sources)
            string(REPLACE "-" "_" target "${folder}_${part}")
            add_executable("${target}" ${sources})
            target_include_directories("${target}" PRIVATE "${directory}")
            if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
                target_compile_options("${target}" PRIVATE -Wall -Wextra -Wpedantic)
            endif()
        endif()
    endforeach()
endforeach()
